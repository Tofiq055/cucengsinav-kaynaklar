package SnakeEyes;


public class SnakeEyes {

    public static void main(String[] args) {

        int die1, die2;   

        int count;  

        count = 0;

        do {
           die1 = (int)(1 + Math.random() * 6);   
           die2 = (int)(1 + Math.random() * 6); 
           count++;                        
           
        } while ( die1 != 1 || die2 != 1 );

        System.out.println("It took " + count + " rolls to get snake eyes [1][1].");

    }  

 }

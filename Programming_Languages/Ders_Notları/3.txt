#include <stdio.h>

int function(int *k)
{
    (*k) += 4;
    return 3 * (*k) - 1;
}

int main()
{
    int i = 10, j = 10, sum1, sum2;

    sum1 = (i / 2) + function(&i);

    sum2 = (j / 2) + function(&j);

    printf("%d %d\n", sum1, sum2);

    return 0;
}

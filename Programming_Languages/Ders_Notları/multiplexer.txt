//as a testbench(left window):

module testbench1; 
reg a, b, c,d,out,sel;
  
  model mux(.a(a), .b(b), .c(c), .d(d),.sel(sel),.out(out) );

initial begin
    
sel<=0;
  a<=$random;
  b<=$random;
  c<=$random;
  d<=$random;
  
end
endmodule

//As a design(right window):

module mux(input a,b,c,d,sel,output reg out);

  always @(a or b or c or d or sel)begin
  case(sel)
    2'b00:out<=a;
    2'b01:out<=b;
    2'b10:out<=c;
    2'b11:out<=d;
  endcase
end
 endmodule

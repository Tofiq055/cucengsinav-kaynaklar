typedef struct LList {	int id;	char name[20];	char surname[20];
	float balance;LList* next;} BankAccount; 
BankAccount *first,*last;
void SetAccount(BankAccount* item,int tid,char* tname, char* tsurname,float tbalance)
{	item->id = tid;
	strcpy(item->name, tname);
	strcpy(item->surname, tsurname);
	item->balance = tbalance;}
void AddItem(int tid,char* tname, char* tsurname,float tbalance)
{
	if(first==NULL)
	{
		first=(BankAccount *)malloc(sizeof(BankAccount));
		SetAccount(first,tid,tname, tsurname,tbalance);
		last=first;
		last->next = NULL;
	}
	else	
	{
		last->next=(BankAccount *)malloc(sizeof(BankAccount));
		last = last->next;
		SetAccount(last,tid,tname, tsurname,tbalance);
		last->next = NULL;
	}
}
